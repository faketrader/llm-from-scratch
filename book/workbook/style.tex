\ifdefined\HCode\else
  \externaldocument[tb-]{../../build/textbook/textbook}
\fi
\ExplSyntaxOn
% 同章题目与解析共同维护；题目出现时生成内部锚点，解析暂存时绑定该锚点，
% 从而无需在章节源码中手填题序或维护题目—解析标签对。
\seq_new:N \g_my_workbook_answers_seq
\int_new:N \g_my_workbook_chapter_int
\int_new:N \g_my_workbook_exercise_int
\bool_new:N \g_my_workbook_answer_pending_bool
\tl_new:N \g_my_workbook_chapter_label_tl
\tl_new:N \g_my_workbook_question_label_tl
\tl_new:N \g_my_workbook_answer_label_tl
\tl_new:N \g_my_workbook_question_level_tl
\RenewDocumentCommand{\optional}{}
  {
    \bool_if:NT \g_my_workbook_answer_pending_bool
      { \tl_gset:Nn \g_my_workbook_question_level_tl { optional } }
    \ifdefined\HCode
      \HCode{<span~class="optional-marker"~role="note"~aria-label="选修内容"><span~aria-hidden="true">⚠</span><span>选修</span></span>}
    \else
      \myoptionalmark
    \fi
  }
\msg_new:nnn { my } { missing-workbook-answer }
  { Each~exercise~must~be~followed~by~one~solution~environment. }
\cs_new_protected:Npn \my_workbook_typeset_answer:nnnn #1#2#3#4
  {
    \ifdefined\HCode
      \HCode{<div~class="my-solution"~data-question="#1"~data-answer="#2"~data-level="#3">}
      \par\noindent\textbf{解析}\quad\ignorespaces#4
      \HCode{</div>}
    \else
      \par\medskip
      \noindent\phantomsection\label{#2}
      \hyperref[#1]{\textbf{\ref*{#1}}}\quad
      \str_if_eq:nnT {#3} { optional } { \myoptionalmark }
      \ignorespaces#4
    \fi
  }
\cs_new_protected:Npn \my_workbook_store_bound_answer:n #1
  {
    \seq_gput_right:Nx \g_my_workbook_answers_seq
      {
        \exp_not:N \my_workbook_typeset_answer:nnnn
          { \exp_not:V \g_my_workbook_question_label_tl }
          { \exp_not:V \g_my_workbook_answer_label_tl }
          { \exp_not:V \g_my_workbook_question_level_tl }
          { \exp_not:n {#1} }
      }
  }
\NewDocumentEnvironment{solution}{+b}
  {
    \bool_if:NTF \g_my_workbook_answer_pending_bool
      {
        \my_workbook_store_bound_answer:n {#1}
        \bool_gset_false:N \g_my_workbook_answer_pending_bool
      }
      { \msg_error:nn { my } { missing-workbook-answer } }
  }
  {}
\NewDocumentCommand{\printsolutions}{}
  {\seq_map_inline:Nn \g_my_workbook_answers_seq {##1}}
\NewDocumentCommand{\workbookchapter}{m}
  {
    \chapter{#1}
    \int_gincr:N \g_my_workbook_chapter_int
    \ifdefined\HCode
      \HCode{<div~class="my-workbook-chapter"~data-chapter="\int_use:N\g_my_workbook_chapter_int"></div>}
    \fi
    \tl_gset:Nx \g_my_workbook_chapter_label_tl
      { my-workbook-chapter-\int_use:N \g_my_workbook_chapter_int }
    \exp_args:NV \label \g_my_workbook_chapter_label_tl
    \seq_gput_right:Nx \g_my_workbook_answers_seq
      {
        \exp_not:N \answerchapter
          { \exp_not:V \g_my_workbook_chapter_label_tl }
      }
  }
\NewDocumentCommand{\tbobject}{m}
  {\ifdefined\HCode\HCode{<span~class="my-textbook-ref"~data-label="#1">（教材对象）</span>}\else（\ref{tb-#1}）\fi}
\NewDocumentEnvironment{exercises}{}
  {\begin{enumerate}[labelwidth=3.2em,leftmargin=4em,labelsep=.6em,align=right]}
  {
    \bool_if:NT \g_my_workbook_answer_pending_bool
      { \msg_error:nn { my } { missing-workbook-answer } }
    \end{enumerate}
  }
\NewDocumentEnvironment{exercise}{}
  {
    \bool_if:NT \g_my_workbook_answer_pending_bool
      { \msg_error:nn { my } { missing-workbook-answer } }
    \int_gincr:N \g_my_workbook_exercise_int
    \tl_gset:Nx \g_my_workbook_question_label_tl
      { my-workbook-question-\int_use:N \g_my_workbook_exercise_int }
    \tl_gset:Nx \g_my_workbook_answer_label_tl
      { my-workbook-answer-\int_use:N \g_my_workbook_exercise_int }
    \tl_gset:Nn \g_my_workbook_question_level_tl { required }
    \renewcommand{\theenumi}{\thechapter.\arabic{enumi}}
    \renewcommand{\labelenumi}
      {\hyperref[\tl_use:N \g_my_workbook_answer_label_tl]{\theenumi.}}
    \item\leavevmode
    \ifdefined\HCode
      \HCode{<div~class="my-exercise"~data-exercise="\int_use:N\g_my_workbook_exercise_int">}
    \fi
    \phantomsection
    \exp_args:NV \label \g_my_workbook_question_label_tl
    \bool_gset_true:N \g_my_workbook_answer_pending_bool
  }
  {\ifdefined\HCode\HCode{</div>}\fi}
\ExplSyntaxOff

% 保留“篇—章”的真实结构，同时将章级目录项排成紧凑、可扫描的列表。
% 该覆盖只在习题入口加载，不影响教材目录。

\makeatother

% 参考解析正文连续排版；章号、标题和公式前缀均取自对应习题章标签。
\RenewDocumentCommand{\answerchapter}{m}{%
  \ifdefined\HCode
    \HCode{<div~class="my-answer-chapter"~data-question-chapter="#1"></div>}%
  \else
    \edef\myanswerchapternumber{\getrefnumber{#1}}%
    \section*{第\myanswerchapternumber{}章\texorpdfstring{\quad}{ }\nameref*{#1}}%
    \phantomsection\label{ans:#1}%
    \addcontentsline{toc}{chapter}{\protect\numberline{第\myanswerchapternumber{}章}\nameref*{#1}}%
    \setcounter{equation}{0}%
    \renewcommand{\theequation}{答\myanswerchapternumber.\arabic{equation}}%
    \renewcommand{\theHequation}{answer.\myanswerchapternumber.\arabic{equation}}%
  \fi}
