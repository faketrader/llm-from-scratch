% Panel 1: ALiBi Penalty Matrix
\node[font=\small\bfseries, text=LFSGray] at (2.2, 4.2) {ALiBi 线性衰减偏置矩阵 ($-\,m \cdot (i-j)$)};

% Draw 5x5 matrix
\begin{scope}[yshift=0.5cm]
  \foreach \i in {0,...,4} {
    \foreach \j in {0,...,4} {
      \pgfmathtruncatemacro{\diff}{\i - \j}
      \ifnum \diff < 0
        % Masked
        \fill[LFSGray!15] (\j*0.9, {-\i*0.7}) rectangle ++(0.8, 0.6);
        \node[font=\tiny, text=LFSGray!60] at ({\j*0.9+0.4}, {-\i*0.7+0.3}) {$-\infty$};
      \else
        \ifnum \diff = 0
          \fill[LFSBlueSoft] (\j*0.9, {-\i*0.7}) rectangle ++(0.8, 0.6);
          \node[font=\scriptsize\bfseries, text=LFSBlue] at ({\j*0.9+0.4}, {-\i*0.7+0.3}) {$0$};
        \else
          \fill[LFSOrangeSoft] (\j*0.9, {-\i*0.7}) rectangle ++(0.8, 0.6);
          \node[font=\scriptsize, text=LFSOrange] at ({\j*0.9+0.4}, {-\i*0.7+0.3}) {$-m\cdot\diff$};
        \fi
      \fi
      \draw[draw=LFSGray!40, line width=0.4pt] (\j*0.9, {-\i*0.7}) rectangle ++(0.8, 0.6);
    }
  }

  % Row labels (Query positions i)
  \foreach \i in {0,...,4} {
    \node[font=\tiny, text=LFSGray, left] at (-0.1, {-\i*0.7+0.3}) {$i=\i$};
  }
  % Col labels (Key positions j)
  \foreach \j in {0,...,4} {
    \node[font=\tiny, text=LFSGray, above] at ({\j*0.9+0.4}, 0.65) {$j=\j$};
  }
\end{scope}
