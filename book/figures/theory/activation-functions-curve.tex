\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex,amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta}
\input{diagram_styles.tex}

\begin{document}
\begin{tikzpicture}[lfs picture]
\begin{axis}[
    width=115mm,
    height=66mm,
    xlabel={输入数值 $x$},
    ylabel={激活值 $\phi(x)$},
    xmin=-3.2, xmax=3.2,
    ymin=-0.6, ymax=3.2,
    grid=both,
    grid style={line width=.2pt, draw=gray!20},
    major grid style={line width=.4pt, draw=gray!35},
    axis lines=middle,
    enlargelimits=false,
    tick label style={font=\footnotesize},
    label style={font=\small},
    legend pos=north west,
    legend style={font=\scriptsize, fill=white, draw=gray!30}
]

% ReLU
\addplot[domain=-3.2:3.2, samples=100, line width=1.1pt, color=LFSGray, dashed] {max(0, x)};
\addlegendentry{ReLU: $\max(0, x)$}

% GELU (tanh 近似)
\addplot[domain=-3.2:3.2, samples=120, line width=1.3pt, color=LFSBlue] {0.5*x*(1 + tanh(sqrt(2/3.14159265)*(x + 0.044715*x^3)))};
\addlegendentry{GELU: $x\Phi(x)$}

% SiLU / Swish
\addplot[domain=-3.2:3.2, samples=120, line width=1.3pt, color=LFSGreen] {x / (1 + exp(-x))};
\addlegendentry{SiLU: $x\cdot\sigma(x)$}

\node[label text, anchor=north] at (axis cs:-1.5, -0.15) {平滑负值下陷谷底};

\end{axis}
\end{tikzpicture}
\end{document}
