\else
  \RequirePackage[noautomatic]{imakeidx}
  % 仅教材生成索引；独立章节复用导言区时不写索引文件。
  \ExplSyntaxOn
  \prop_new:N \g_my_index_sort_prop
  \prop_new:N \g_my_index_seen_prop
  \int_new:N \g_my_index_anchor_int
  \msg_new:nnn {my} {missing-index-key}
    {Missing~pinyin~sort~key~for~term:~#1.}
  \NewDocumentCommand{\DeclareTermIndexKey}{mm}
    {\prop_gput:Nnn \g_my_index_sort_prop {#1} {#2}}
  \NewDocumentCommand{\TermIndexPage}{mm}{\hyperlink{#1}{#2}}
  \cs_new_protected:Npn \my_write_index:nn #1#2
    {\index[#1]{#2}}
  \cs_new_protected:Npn \my_index_english:nn #1#2
    {
      \tl_set:Nx \l_tmpa_tl
        {\tl_to_str:n {#1} @\exp_not:n {#1}|TermIndexPage{#2}}
      \exp_args:NnV \my_write_index:nn {en} \l_tmpa_tl
    }
  \cs_new_protected:Npn \my_index_term:nnn #1#2#3
    {
      \prop_get:NnNTF \g_my_index_sort_prop {#1} \l_tmpa_tl
        {
          \tl_set:Nx \l_tmpb_tl {\thechapter / \tl_to_str:n {#1}}
          \exp_args:NNV \prop_if_in:NnF \g_my_index_seen_prop \l_tmpb_tl
            {
              \exp_args:NNV \prop_gput:Nnn \g_my_index_seen_prop \l_tmpb_tl {}
              \int_gincr:N \g_my_index_anchor_int
              \tl_set:Nx \l_tmpb_tl {my-term-\int_use:N \g_my_index_anchor_int}
              \exp_args:NV \hypertarget \l_tmpb_tl {}
              \tl_set:Nx \l_tmpc_tl
                {\l_tmpa_tl @\exp_not:n {#1}|TermIndexPage{\l_tmpb_tl}}
              \exp_args:NnV \my_write_index:nn {cn} \l_tmpc_tl
              \exp_args:NnV \my_index_english:nn {#2} \l_tmpb_tl
            }
        }
        {\msg_error:nnn {my} {missing-index-key} {#1}}
    }
  \ifSubfilesClassLoaded{}
    {
      % 主入口已关闭 hyperref 默认的整页链接包装，改用术语锚点。
      \makeindex[name=cn,title=中文术语索引,intoc,columns=2]
      \makeindex[name=en,title=英文术语索引,intoc,columns=2]
      \cs_new_eq:NN \my_index_original_term:nnn \term
      \RenewDocumentCommand{\term}{mmm}
        {
          \my_index_term:nnn {#1}{#2}{#3}
          \exp_args:Nne \regex_match:nnTF
            { \A ( \$ | [A-Za-z0-9] ) }
            { \tl_to_str:n {#1} }
            { \CJKecglue }
            { \CJKglue }
          \my_index_original_term:nnn {#1}{#2}{#3}
        }
    }
  \ExplSyntaxOff
\fi
